const candidate = String(value ?? '');
if (!candidate || /[\u0000-\u0020\u007f"'()\\<>{};*`]|[?#]/u.test(candidate)) return '';
    if (/%(?:0[0-9a-f]|1[0-9a-f]|7f|22|27|28|29|3b|3c|3e|5c|60|7b|7d)/iu.test(candidate)) return '';
    if (/^https?:\/\//iu.test(candidate)) {
      try {
        if (!['http:', 'https:'].includes(new URL(candidate).protocol)) return '';
      } catch (_error) {
        return '';
      }
    } else if (!(candidate.startsWith('./')
      || candidate.startsWith('../')
      || (candidate.startsWith('/') && !candidate.startsWith('//')))) return '';
    return candidate.endsWith('/') ? candidate : `${candidate}/`;
  }

  // 注册表里的头像路径相对于 avatarBase（即 profile photo/），允许带子目录，例如 profile photo/gr/凯莎.png。
  function avatarPath(character) {
    const raw = String(character.avatar || `${character.canonicalName || ''}.png`).replace(/\\/gu, '/');
    const segments = raw
      .replace(/^(?:\.\/)?profile photo\//u, '')
      .split('/')
      .filter((segment) => segment && segment !== '.' && segment !== '..');
    if (!segments.length || segments.at(-1) === '.png') return '';
    return segments.map(encodeURIComponent).join('/');
  }

  function avatarUrl(character) {
    const relativePath = avatarPath(character);
    return relativePath && runtimeAvatarBase ? `${runtimeAvatarBase}${relativePath}` : '';
  }

  function hexToRgb(hex) {
    const normalized = String(hex || '').replace('#', '');
    if (!/^[0-9A-F]{6}$/iu.test(normalized)) return [117, 107, 97];
    return [0, 2, 4].map((offset) => Number.parseInt(normalized.slice(offset, offset + 2), 16));
  }

  function mixWithWhite(hex, amount) {
    const values = hexToRgb(hex).map((value) =>
      Math.round(value + (255 - value) * amount),
    );
    return `#${values.map((value) => value.toString(16).padStart(2, '0')).join('')}`;
  }

  function boundedMaterialNumber(value, fallback, minimum, maximum) {
    const number = Number(value);
    return Number.isInteger(number) && number >= minimum && number <= maximum
      ? number
      : fallback;
  }

  function applyBubbleMaterial(element, character) {
    const source = character?.bubbleStyle;

    const angle = boundedMaterialNumber(source?.angle, BUBBLE_DEFAULT.angle, 0, 179);
    const step = boundedMaterialNumber(source?.step, BUBBLE_DEFAULT.step, 12, 32);
    const radius = boundedMaterialNumber(source?.radius, BUBBLE_DEFAULT.radius, 4, 24);
    const grain = boundedMaterialNumber(source?.grain, BUBBLE_DEFAULT.grain, 2, 12);

    const reporter = `<script>(function(){var t=${JSON.stringify(ticket)};function s(){try{parent.postMessage({lzEmbed:t,height:Math.max(document.documentElement.scrollHeight,document.body?document.body.scrollHeight:0)},'*')}catch(e){}}addEventListener('load',s);if(window.ResizeObserver){new ResizeObserver(s).observe(document.documentElement)}setTimeout(s,60);setInterval(s,1500)})()<\/script>`;
    frame.srcdoc = `<!doctype html><html lang="zh-CN"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><style>html{color-scheme:${colorScheme}}html,body{margin:0;padding:0;background:transparent}body{overflow:hidden}img,video{max-width:100%}</style></head><body>${token.html}${reporter}</body></html>`;

  function avatarStorageKey() {
    return `${STORAGE_PREFIX}player-avatar.${encodeURIComponent(playerName || 'default')}`;
  }

  function openAvatarDatabase() {
    return new Promise((resolve, reject) => {
      if (!globalThis.indexedDB) {
        reject(new Error('IndexedDB unavailable'));
        return;
      }
      let settled = false;
      let timer = 0;
      let request;
      const finish = (kind, value) => {
        if (settled) return;
        settled = true;
        globalThis.clearTimeout(timer);
        if (kind === 'resolve') resolve(value);
        else reject(value);
      };
      try {
        request = globalThis.indexedDB.open(AVATAR_DATABASE, 1);
      } catch (error) {
        finish('reject', error);
        return;
      }
      timer = globalThis.setTimeout(() => {
        finish('reject', new Error('IndexedDB open timed out'));
      }, AVATAR_DATABASE_TIMEOUT);
      request.addEventListener('upgradeneeded', () => {
        if (!request.result.objectStoreNames.contains(AVATAR_STORE)) {
          request.result.createObjectStore(AVATAR_STORE, { keyPath: 'name' });
        }
      });
      request.addEventListener('success', () => {
        if (settled) {
          request.result.close();
          return;
        }
        finish('resolve', request.result);
      }, { once: true });
      request.addEventListener('error', () => finish('reject', request.error || new Error('IndexedDB open failed')), { once: true });
      request.addEventListener('blocked', () => finish('reject', new Error('IndexedDB open blocked')), { once: true });
    });
  }

  function runAvatarTransaction(database, mode, operation) {
    return new Promise((resolve, reject) => {
      let settled = false;
      let timer = 0;
      let requestResult = null;
      let transaction;
      const finish = (kind, value) => {
        if (settled) return;
        settled = true;
        globalThis.clearTimeout(timer);
        if (kind === 'resolve') resolve(value);
        else reject(value);
      };
      try {
        transaction = database.transaction(AVATAR_STORE, mode);
        const request = operation(transaction.objectStore(AVATAR_STORE));
        request?.addEventListener('success', () => {
          requestResult = request.result;
        }, { once: true });
        request?.addEventListener('error', () => {
          finish('reject', request.error || new Error('IndexedDB request failed'));
        }, { once: true });
      } catch (error) {
        finish('reject', error);
        return;
      }
      timer = globalThis.setTimeout(() => {
        try {
          transaction.abort();
        } catch (_error) {
          // A transaction that already ended needs no further cleanup.
        }
        finish('reject', new Error('IndexedDB transaction timed out'));
      }, AVATAR_DATABASE_TIMEOUT);
      transaction.addEventListener('complete', () => finish('resolve', requestResult), { once: true });
      transaction.addEventListener('error', () => finish('reject', transaction.error || new Error('IndexedDB transaction failed')), { once: true });
      transaction.addEventListener('abort', () => finish('reject', transaction.error || new Error('IndexedDB transaction aborted')), { once: true });
    });
  }

  async function readAvatarDatabaseRecord() {
    let database = null;
    try {
      database = await openAvatarDatabase();
      return await runAvatarTransaction(database, 'readonly', (store) => store.get(playerName));
    } catch (_error) {
      return null;
    } finally {
      database?.close();
    }
  }

  function readAvatarLocalRecord() {
    try {
      return globalThis.localStorage?.getItem(avatarStorageKey()) || null;
    } catch (_error) {
      return null;
    }
  }

  async function readPlayerAvatar() {
    if (!playerName) return null;
    const [databaseValue, localValue] = await Promise.all([
      readAvatarDatabaseRecord(),
      Promise.resolve(readAvatarLocalRecord()),
    ]);
    const databaseRecord = core.normalizeAvatarRecord(databaseValue, playerName);
    const localRecord = core.normalizeAvatarRecord(localValue, playerName);
    lastAvatarStamp = Math.max(
      lastAvatarStamp,
      databaseRecord?.updatedAt || 0,
      localRecord?.updatedAt || 0,
    );
    return core.avatarDataFromRecord(
      core.chooseLatestAvatarRecord(databaseRecord, localRecord),
    );
  }

  async function writeAvatarDatabaseRecord(record) {
    let database = null;
    try {
      database = await openAvatarDatabase();
      await runAvatarTransaction(database, 'readwrite', (store) => store.put(record));
      return true;
    } catch (_error) {
      return false;
    } finally {
      database?.close();
    }
  }

  function writeAvatarLocalRecord(record) {
    try {
      globalThis.localStorage?.setItem(avatarStorageKey(), JSON.stringify(record));
      return true;
    } catch (_error) {
      return false;
    }
  }

  function nextAvatarStamp() {
    lastAvatarStamp = Math.max(Date.now(), lastAvatarStamp + 1);
    return lastAvatarStamp;
  }

  async function persistAvatarRecord(record) {
    const [databaseStored, localStored] = await Promise.all([
      writeAvatarDatabaseRecord(record),
      Promise.resolve(writeAvatarLocalRecord(record)),
    ]);
    return databaseStored || localStored;
  }

  function makeAvatarRecord(dataUrl, deleted) {
    return {
      name: playerName,
      dataUrl,
      updatedAt: nextAvatarStamp(),
      deleted,
    };
  }

  function writePlayerAvatar(dataUrl) {
    return persistAvatarRecord(makeAvatarRecord(dataUrl, false));
  }

  function deletePlayerAvatar() {
    return persistAvatarRecord(makeAvatarRecord(null, true));
  }

  function loadImage(source) {
    return new Promise((resolve, reject) => {
      const image = new Image();
      image.addEventListener('load', () => resolve(image), { once: true });
      image.addEventListener('error', () => reject(new Error('无法读取图片')), { once: true });
      image.src = source;
    });
  }

  async function resizeAvatar(file) {
    if (!file || !AVATAR_TYPES.has(file.type)) {
      throw new Error('仅支持 PNG、JPEG 或 WebP 图片');
    }
    if (file.size > AVATAR_FILE_LIMIT) throw new Error('图片不能超过 8MB');
    const header = new Uint8Array(
      await file.slice(0, Math.min(file.size, IMAGE_HEADER_LIMIT)).arrayBuffer(),
    );
    const metadata = core.validateAvatarImageHeader(header, file.type, IMAGE_PIXEL_LIMIT);
    if (!globalThis.URL?.createObjectURL || !globalThis.URL?.revokeObjectURL) {
      throw new Error('当前浏览器无法安全读取本地图片');
    }
    const objectUrl = globalThis.URL.createObjectURL(file);
    try {
      const image = await loadImage(objectUrl);
      if (image.naturalWidth !== metadata.width || image.naturalHeight !== metadata.height) {
        throw new Error('图片头信息与解码尺寸不一致');
      }
      const sourceSize = Math.min(image.naturalWidth, image.naturalHeight);
      const sourceX = (image.naturalWidth - sourceSize) / 2;
      const sourceY = (image.naturalHeight - sourceSize) / 2;
      const canvas = document.createElement('canvas');
      canvas.width = AVATAR_EDGE;
      canvas.height = AVATAR_EDGE;
      const context = canvas.getContext('2d', { alpha: false });
      if (!context) throw new Error('当前浏览器无法处理图片');
      context.fillStyle = '#E9E3D8';
      context.fillRect(0, 0, AVATAR_EDGE, AVATAR_EDGE);
      context.drawImage(
        image,
        sourceX,
        sourceY,
        sourceSize,
        sourceSize,
        0,
        0,
        AVATAR_EDGE,
        AVATAR_EDGE,
      );
      return canvas.toDataURL('image/webp', 0.86);
    } finally {
      globalThis.URL.revokeObjectURL(objectUrl);
    }
  }

  function updatePlayerAvatarStatus() {
    const count = Object.keys(state.avatarOverrides).length;
    setText(elements.avatarStatus, count ? `已自定义 ${count} 位角色` : '点击正文中的名字即可更换');
    if (elements.avatarReset) elements.avatarReset.disabled = count === 0 && !state.playerAvatar;
  }

  function bindAvatarControls() {
    elements.avatarInput?.addEventListener('change', async () => {
      const [file] = elements.avatarInput.files || [];
      elements.avatarInput.value = '';
      if (!file) return;
      const target = state.editingAvatar || playerName;
      if (!target) return;
      const revision = avatarRevision.begin();
      try {
        await enqueueAvatarMutation(async () => {
          const dataUrl = await resizeAvatar(file);
          if (!avatarRevision.isCurrent(revision)) return;
          writeAvatarOverride(target, { dataUrl });
          if (isPlayerSpeaker(target)) {
            state.playerAvatar = dataUrl;
            await writePlayerAvatar(dataUrl).catch(() => false);
          }
          rerenderAfterAvatarChange();
          closeAvatarEditor();
          showToast(`${target} 的头像已保存在本机`);
        });
      } catch (error) {
        if (avatarRevision.isCurrent(revision)) {
          showToast(error instanceof Error ? error.message : '头像处理失败');
        }
      }
    });
    elements.avatarReset?.addEventListener('click', async () => {
      const revision = avatarRevision.begin();
      try {
        await enqueueAvatarMutation(async () => {
          for (const name of Object.keys(state.avatarOverrides)) writeAvatarOverride(name, null);
          if (playerName && state.playerAvatar) {
            await deletePlayerAvatar().catch(() => false);
            state.playerAvatar = null;
          }
          if (!avatarRevision.isCurrent(revision)) return;
          rerenderAfterAvatarChange();
          showToast('已恢复全部默认头像');
        });
      } catch (error) {
        if (avatarRevision.isCurrent(revision)) {
          showToast(error instanceof Error ? error.message : '头像重置失败');
        }
      }
    });
    // 小手机等同源页面改了头像时实时跟进。
    globalThis.addEventListener?.('storage', (event) => {
      if (event.key !== AVATAR_OVERRIDES_KEY) return;
      state.avatarOverrides = readAvatarOverrides();
      rerenderAfterAvatarChange();
    });
  }

  let visibilityObserver = null;
  function observeAnimatedElements() {
    visibilityObserver?.disconnect();
    const targets = root.querySelectorAll('.lz-dialogue, .lz-observed-effect');
    if (!('IntersectionObserver' in globalThis)) {
      targets.forEach((target) => target.classList.add('is-visible', 'is-in-view'));
      return;
    }
    visibilityObserver = new IntersectionObserver(
      (entries) => {
        for (const entry of entries) {
          entry.target.classList.toggle('is-in-view', entry.isIntersecting);
          if (entry.isIntersecting && entry.target.classList.contains('lz-observed-effect')) {
            entry.target.classList.add('is-visible');
          }
        }
      },
      { rootMargin: '10% 0px 10%', threshold: 0.08 },
    );
    targets.forEach((target) => visibilityObserver.observe(target));
  }

  async function initialize() {
    renderMetadata();
    applySettings();
    bindSettings();
    renderContent();
    bindAvatarControls();
    updatePlayerAvatarStatus();
    observeAnimatedElements();
    const hydrationRevision = avatarRevision.current();
    const storedAvatar = await readPlayerAvatar();
    if (!avatarRevision.isCurrent(hydrationRevision) || !storedAvatar) return;
    state.playerAvatar = storedAvatar;
    renderContent();
    updatePlayerAvatarStatus();
    observeAnimatedElements();
  }

  initialize().catch(() => {
    renderContent();
    applySettings();
  });
